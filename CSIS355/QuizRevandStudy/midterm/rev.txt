const os = require("os");
const path = require("path");
const http = require("http");
const fs = require("fs");

filepath = path.join(__dirname, "testfile.txt");
fs.writeFile(filepath, "content", (err)=>{
    if(err) throw err;
    console.log("file made");
});

fs.appendFile(filepath, "more content", (err)=>{
    if(err) throw err;
    console.log("file made");
});

const server = http.createServer((req, res)=>{
    if(req.url === "/"){
        fs.readFile(filepath, 'utf-8', (err, content)=>{
            if(err) throw err;
            res.writeHead(200, {'content-type':'text/text'});
            res.end(content);
        });
    }else{
        res.writeHead(404);
        res.end();
    }
});

function student(name, gpa){
    this.name = name;
    this.gpa = gpa;
    this.displayInfo = ()=>{
        console.log("Name: " + this.name + " GPA: " + this.gpa);
    }
}

var stu = new student("john", 4.0);
stu.displayInfo();
